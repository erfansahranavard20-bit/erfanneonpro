async function api(url, options = {}) {
  const headers = { ...(options.body ? {'Content-Type':'application/json'} : {}), ...(options.headers || {}) };
  if (options.method && options.method !== 'GET' && csrfToken) headers['X-CSRF-Token'] = csrfToken;
  const response = await fetch(url, {...options, headers, credentials:'same-origin'});
  let data = {};
  try { data = await response.json(); } catch {}
  if (!response.ok) throw new Error(data.error || 'درخواست انجام نشد.');
  return data;
}

    }).format(now);

    }).format(now);
  }
  updateClock();
  setInterval(updateClock, 1000);

  const storageKey = 'erfan-neon-dashboard-tasks-v1';
  let tasks = [];
  try {
    const saved = JSON.parse(localStorage.getItem(storageKey) || '[]');
    if (Array.isArray(saved)) {
      tasks = saved.filter(t =>
        t && typeof t.id === 'string' &&
        typeof t.text === 'string' &&
        typeof t.done === 'boolean'
      ).slice(0, 200);
    }
  } catch {}

  function saveTasks() {
    try {
      localStorage.setItem(storageKey, JSON.stringify(tasks));
    } catch {
      showToast('ذخیره در مرورگر انجام نشد.');
    }
  }

  function renderTasks() {
    const list = byId('taskList');

      new Intl.NumberFormat('fa-IR').format(pending) + ' باقی‌مانده';

    list.innerHTML = tasks.map(task => `
      <div class="task-row ${task.done ? 'is-done' : ''}">

            ${task.done ? 'checked' : ''}>

        </label>

      reactionButton.classList.remove('is-ready');

      reactionMessage.textContent = '⚡ زمان واکنش: ' + elapsed + ' میلی‌ثانیه';
      return;
    }
    reactionState = 'waiting';
    reactionButton.classList.remove('is-ready');
    reactionButton.classList.add('is-waiting');
    reactionButton.textContent = 'صبر کن...';
    reactionMessage.textContent = 'هنوز نه!';
    reactionTimeout = setTimeout(() => {
      reactionState = 'ready';
      reactionStartedAt = performance.now();
      reactionButton.classList.remove('is-waiting');
      reactionButton.classList.add('is-ready');
      reactionButton.textContent = 'حالا لمس کن!';
      reactionMessage.textContent = 'همین الان!';
    }, 1200 + Math.random() * 2800);
  });

  // Tic-tac-toe
  const cells = Array.from(root.querySelectorAll('[data-cell]'));
  const ticMessage = document.getElementById('arcadeTicMessage');
  const wins = [
    [0,1,2],[3,4,5],[6,7,8],[0,3,6],
    [1,4,7],[2,5,8],[0,4,8],[2,4,6]
  ];
  let board = Array(9).fill('');
  let ticOver = false;
  let ticBusy = false;

  function winner(mark) {
    return wins.some(line => line.every(i => board[i] === mark));
  }

  function renderBoard() {
    cells.forEach((cell, i) => {
      cell.textContent = board[i];
      cell.disabled = ticOver || ticBusy || board[i] !== '';
      cell.classList.toggle('mark-x', board[i] === '❌');
      cell.classList.toggle('mark-o', board[i] === '⭕');
    });
  }

  function finishIfNeeded() {
    if (winner('❌')) {
      ticMessage.textContent = '🎉 تو بردی!';
      ticOver = true;
    } else if (winner('⭕')) {
      ticMessage.textContent = 'کامپیوتر این دور رو برد!';
      ticOver = true;
    } else if (board.every(Boolean)) {
      ticMessage.textContent = 'مساوی شد! 🤝';
      ticOver = true;
    } else {
      return false;
    }
    renderBoard();
    return true;
  }

  function computerMove() {
    // Win if possible, otherwise block the player, then prefer center/corners.
    function findWinningMove(mark) {
      for (let i = 0; i < 9; i++) {
        if (board[i]) continue;
        board[i] = mark;
        const winsNow = winner(mark);
        board[i] = '';
        if (winsNow) return i;
      }
      return -1;
    }

    let move = findWinningMove('⭕');
    if (move < 0) move = findWinningMove('❌');
    if (move < 0 && !board[4]) move = 4;
    if (move < 0) {
      const corners = [0,2,6,8].filter(i => !board[i]);
      if (corners.length) move = corners[Math.floor(Math.random() * corners.length)];
    }
    if (move < 0) {
      const empty = board.map((v,i) => v ? -1 : i).filter(i => i >= 0);
      if (empty.length) move = empty[Math.floor(Math.random() * empty.length)];
    }
    if (move >= 0) board[move] = '⭕';
  }

  cells.forEach((cell, i) => cell.addEventListener('click', () => {
    if (ticOver || ticBusy || board[i]) return;
    board[i] = '❌';
    renderBoard();
    if (finishIfNeeded()) return;
    ticBusy = true;
    renderBoard();
    setTimeout(() => {
      computerMove();
      ticBusy = false;
      if (!finishIfNeeded()) ticMessage.textContent = 'نوبت توئه!';
      renderBoard();
    }, 300);
  }));

  document.getElementById('arcadeTicReset').addEventListener('click', () => {
    board = Array(9).fill('');
    ticOver = false;
    ticBusy = false;
    ticMessage.textContent = 'نوبت توئه!';
    renderBoard();
  });

  renderBoard();
})();

        if (fresh) { paint(key === '.' ? '0.' : key); fresh = false; }
        else if (key === '.') { if (!current.includes('.')) paint(current + '.'); }
        else paint(current === '0' ? key : current + key);
        return;
      }
      if ('+-*/'.includes(key)) {
        if (op && !fresh) calculate();
        stored = Number(current);
        op = key; fresh = true; return;
      }
      if (key === '=' && op && stored !== null) calculate();

      const b = Number(current), a = stored;

      if (op === '-') result = a - b;

      if (op === '/') result = b === 0 ? NaN : a / b;

'use strict';
const form = document.getElementById('adminLoginForm');
const message = document.getElementById('message');
let csrfToken = '';
async function init() {
  try {
    const response = await fetch('/api/bootstrap', { cache: 'no-store' });
    if (!response.ok) throw new Error('دریافت اطلاعات ورود ناموفق بود.');
    const data = await response.json();
    csrfToken = data.csrfToken;
    if (data.authenticated) location.replace('/');
  } catch (error) {
    message.textContent = error.message || 'ارتباط با سرور برقرار نشد.';
  }
}
form.addEventListener('submit', async event => {
  event.preventDefault();
  const button = form.querySelector('button[type="submit"]');
  button.disabled = true;
  message.textContent = 'در حال بررسی…';
  try {
    if (!csrfToken) throw new Error('صفحه را تازه‌سازی کن و دوباره تلاش کن.');
    const response = await fetch('/api/login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'x-csrf-token': csrfToken },
      body: JSON.stringify({ username: form.elements.username.value, password: form.elements.password.value }),
      cache: 'no-store'
    });
    const data = await response.json();
    if (!response.ok) throw new Error(data.error || 'ورود ناموفق بود.');
    location.replace('/');
  } catch (error) {
    message.textContent = error.message || 'خطا در ارتباط با سرور.';
  } finally {
    button.disabled = false;
  }
});
init();

(() => {
  'use strict';

  const root = document.getElementById('erfan-nexus');
  if (!root || root.dataset.ready) return;
  root.dataset.ready = '1';

  const $ = (selector) => root.querySelector(selector);
  const result = $('#nx-result');
  const input = $('#nx-input');
  const xpKey = 'erfanNexusXP_v1';
  const missionKey = 'erfanNexusMissions_v1';

  const readNumber = (key) => {
    try { return Math.max(0, Number(localStorage.getItem(key)) || 0); }
    catch { return 0; }
  };
  const readMissions = () => {
    try {
      const data = JSON.parse(localStorage.getItem(missionKey) || '{}');
      return data && typeof data === 'object' ? data : {};
    } catch { return {}; }
  };
  const say = (message) => {
    result.textContent = message;
  };
  const normalize = (value) => String(value || '')
    .toLowerCase()
    .replace(/[ي]/g, 'ی')
    .replace(/[ك]/g, 'ک')
    .replace(/\s+/g, ' ')
    .trim();

  function award(id, points = 5) {
    const missions = readMissions();
    if (missions[id]) return;
    missions[id] = true;
    try {
      localStorage.setItem(missionKey, JSON.stringify(missions));
      localStorage.setItem(xpKey, String(readNumber(xpKey) + points));
    } catch {}
    updateStats();
  }

  function updateStats() {
    $('#nx-xp').textContent = readNumber(xpKey);
    $('#nx-count').textContent = Object.keys(readMissions()).length;
    root.querySelectorAll('[data-mission]').forEach((node) => {
      const done = !!readMissions()[node.dataset.mission];
      node.classList.toggle('nx-done', done);
      const status = node.querySelector('.nx-status');
      if (status) status.textContent = done ? 'انجام شد ✓' : 'باز';
    });
  }

  function findTarget(command) {
    const all = [...document.querySelectorAll('main section, body > section, #about, #projects, #contact, #arcade, #tools, #dashboard')];
    const terms = [
      { names: ['بازی', 'بازی ها', 'بازی‌ها', 'دوز', 'حدس عدد', 'واکنش'], ids: ['arcade', 'games'], words: ['بازی', 'دوز', 'حدس عدد', 'واکنش'] },
      { names: ['ابزار', 'ابزارها', 'ماشین حساب', 'ماشین‌حساب', 'تبدیل واحد', 'کرنومتر', 'تایمر'], ids: ['tools', 'tool-center'], words: ['ابزار', 'ماشین حساب', 'ماشین‌حساب', 'تبدیل واحد', 'کرنومتر', 'تایمر'] },
      { names: ['پروژه', 'پروژه ها', 'پروژه‌ها'], ids: ['projects'], words: ['پروژه'] },
      { names: ['درباره', 'درباره من'], ids: ['about'], words: ['درباره'] },
      { names: ['تماس', 'ارتباط'], ids: ['contact'], words: ['تماس', 'ارتباط'] },
      { names: ['داشبورد'], ids: ['dashboard'], words: ['داشبورد'] }
    ];

    for (const group of terms) {
      if (!group.names.some(name => command.includes(normalize(name)))) continue;
      for (const id of group.ids) {
        const found = document.getElementById(id);
        if (found) return found;
      }
      const byText = all.find(el => group.words.some(word => normalize(el.innerText).includes(word)));
      if (byText) return byText;
    }
    return null;
  }

  function runCommand(raw) {
    const command = normalize(raw);
    if (!command) {
      say('یک فرمان بنویس؛ مثلاً «بازی‌ها رو باز کن».');
      return;
    }

    award('first-command', 10);

    if (/^(راهنما|کمک|چه فرمان|دستورها)/.test(command)) {
      say('فرمان‌های نمونه: «بازی‌ها رو باز کن»، «ابزارها»، «پروژه‌ها»، «درباره من»، «تماس» یا یک عبارت برای جست‌وجو.');
      return;
    }

    if (command.includes('امتیاز') || command.includes('ماموریت') || command.includes('مأموریت')) {
      $('#nx-missions').scrollIntoView({ behavior: 'smooth', block: 'center' });
      say(`امتیاز فعلی: ${readNumber(xpKey)}. مأموریت‌ها در همین بخش نمایش داده می‌شوند.`);
      return;
    }

    if (command.includes('حالت تمرکز') || command.includes('تمرکز')) {
      root.classList.toggle('nx-focus');
      say('حالت تمرکز برای پنل Nexus تغییر کرد.');
      return;
    }

    const target = findTarget(command);
    if (target && target !== root) {
      target.scrollIntoView({ behavior: 'smooth', block: 'start' });
      award('explorer', 15);
      say('بخش موردنظر پیدا شد و به آن منتقل شدی.');
      return;
    }

    const searchable = [...document.querySelectorAll('main section, body > section, article, .project-card, .tool-card')];
    const match = searchable.find(el => {
      const text = normalize(el.innerText);
      return text && command.split(' ').filter(word => word.length > 2).some(word => text.includes(word));
    });
    if (match) {
      match.scrollIntoView({ behavior: 'smooth', block: 'center' });
      award('search', 10);
      say('یک بخش مرتبط پیدا شد. اگر این همان چیزی نیست که می‌خواستی، عبارت دقیق‌تری بنویس.');
      return;
    }

    say('این فرمان رو هنوز نمی‌شناسم. «راهنما» رو بنویس تا فرمان‌های موجود رو ببینی.');
  }

  $('#nx-run').addEventListener('click', () => runCommand(input.value));
  input.addEventListener('keydown', (event) => {
    if (event.key === 'Enter') runCommand(input.value);
  });

  root.querySelectorAll('[data-command]').forEach((button) => {
    button.addEventListener('click', () => runCommand(button.dataset.command));
  });

  $('#nx-search').addEventListener('click', () => {
    const query = normalize(input.value);
    if (!query) {
      say('اول کلمه‌ای برای جست‌وجو بنویس.');
      input.focus();
      return;
    }
    const elements = [...document.querySelectorAll('body *')].filter(el =>
      el.children.length === 0 && el.textContent && normalize(el.textContent).includes(query)
    );
    const match = elements.find(el => !root.contains(el));
    if (match) {
      match.scrollIntoView({ behavior: 'smooth', block: 'center' });
      award('search', 10);
      say('نتیجه پیدا شد و به آن منتقل شدی.');
    } else {
      say('نتیجه‌ای پیدا نشد. عبارت کوتاه‌تر یا متفاوتی امتحان کن.');
    }
  });

  $('#nx-voice').addEventListener('click', () => {
    const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!SpeechRecognition) {
      say('تشخیص گفتار در این مرورگر پشتیبانی نمی‌شود. فرمان رو تایپ کن.');
      return;
    }
    const recognition = new SpeechRecognition();
    recognition.lang = 'fa-IR';
    recognition.interimResults = false;
    recognition.maxAlternatives = 1;
    recognition.onresult = (event) => {
      input.value = event.results[0][0].transcript;
      runCommand(input.value);
    };
    recognition.onerror = () => say('تشخیص صدا انجام نشد؛ اجازهٔ میکروفون و پشتیبانی مرورگر رو بررسی کن.');
    recognition.start();
    say('گوش می‌کنم… یک فرمان کوتاه بگو.');
  });

  $('#nx-reset').addEventListener('click', () => {
    if (!window.confirm('امتیازها و مأموریت‌های ذخیره‌شده روی همین مرورگر پاک شوند؟')) return;
    try {
      localStorage.removeItem(xpKey);
      localStorage.removeItem(missionKey);
    } catch {}
    updateStats();
    say('پیشرفت محلی پاک شد.');
  });

  updateStats();
})();

(() => {
  'use strict';
  const root = document.getElementById('erfan-universe');
  if (!root || root.dataset.ready) return;
  root.dataset.ready = '1';

  const $ = selector => root.querySelector(selector);
  const canvas = $('#u-canvas');
  const ctx = canvas.getContext('2d');
  const status = $('#u-status');
  const xpKey = 'erfanUniverseXP_v1';
  const visitedKey = 'erfanUniverseVisited_v1';

  const planets = [
    { id: 'arcade', name: 'سیارهٔ بازی‌ها', color: '#62e7ff',
      desc: 'دروازهٔ سرگرمی و چالش‌ها', target: '#arcade' },
    { id: 'tools', name: 'سیارهٔ ابزارها', color: '#bd8cff',
      desc: 'ابزارهای کاربردی سایت', target: '#tools' },
    { id: 'projects', name: 'سیارهٔ پروژه‌ها', color: '#68ffc1',
      desc: 'پروژه‌ها و دستاوردها', target: '#projects' }
  ];

  function readVisited() {
    try {
      const data = JSON.parse(localStorage.getItem(visitedKey) || '[]');
      return Array.isArray(data) ? data : [];
    } catch { return []; }
  }
  function getXP() {
    try { return Math.max(0, Number(localStorage.getItem(xpKey)) || 0); }
    catch { return 0; }
  }
  function saveVisit(id) {
    try {
      const visited = readVisited();
      if (!visited.includes(id)) {
        visited.push(id);
        localStorage.setItem(visitedKey, JSON.stringify(visited));
        localStorage.setItem(xpKey, String(getXP() + 10));
      }
    } catch {}
    updateStats();
  }
  function updateStats() {
    $('#u-xp').textContent = getXP();
    $('#u-visited').textContent = readVisited().length;
    root.querySelectorAll('[data-planet]').forEach(button => {
      const visited = readVisited().includes(button.dataset.planet);
      button.setAttribute('aria-pressed', String(visited));
    });
  }

  let width = 0, height = 0, dpr = 1, stars = [], frame = 0;
  function resize() {
    const rect = canvas.getBoundingClientRect();
    width = Math.max(1, rect.width);
    height = Math.max(1, rect.height);
    dpr = Math.min(window.devicePixelRatio || 1, 2);
    canvas.width = Math.round(width * dpr);
    canvas.height = Math.round(height * dpr);
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    stars = Array.from({ length: Math.min(110, Math.floor(width * height / 1700)) },
      (_, i) => ({
        x: (i * 73.7 % 1000) / 1000 * width,
        y: (i * 137.3 % 1000) / 1000 * height,
        r: 0.4 + (i % 4) * 0.35,
        phase: i * 0.7
      }));
    draw();
  }

  function draw() {
    ctx.clearRect(0, 0, width, height);
    stars.forEach(star => {
      const glow = 0.35 + 0.65 * (0.5 + 0.5 * Math.sin(frame * 0.025 + star.phase));
      ctx.beginPath();
      ctx.fillStyle = `rgba(190,235,255,${glow})`;
      ctx.arc(star.x, star.y, star.r, 0, Math.PI * 2);
      ctx.fill();
    });

    const cx = width * 0.5, cy = height * 0.5;
    const scale = Math.min(width / 340, height / 230, 1.1);

========== FILE: ./README.md ==========

# ERFAN Neon Portfolio

A responsive Persian personal portfolio with neon styling, project search/filtering, server-side admin authentication, and a JSON-file-backed project manager (compatible with Android Termux).

## Requirements
- Node.js 20 or newer
- npm

## Run locally
1. Extract this folder.
2. Copy `.env.example` to `.env`.
3. Replace `SESSION_SECRET` with a random secret of at least 32 characters. Replace `ADMIN_PASSWORD` with a unique password of at least 12 characters.
4. Run `npm install`
5. Run `npm start`
6. Open `http://localhost:3000`
7. Select **ورود مدیریت** and use the username/password from `.env`.

Generate a strong secret with one of these commands:
- macOS/Linux: `openssl rand -hex 32`
- Node: `node -e "console.log(require('crypto').randomBytes(32).toString('hex'))"`

## What is included
- Responsive Persian portfolio and animated neon visual style
- Project search and category filters
- Admin-only project creation and deletion persisted in `projects.json`
- Server-side session cookie (`HttpOnly`, `SameSite=Lax`), session rotation after login
- CSRF token checks for state-changing requests
- Login rate limiting, Helmet security headers, parameterized SQL queries
- No public self-signup: the admin account is configured by the site owner through environment variables

## Production / custom domain
This archive does not itself publish a website or purchase a domain. To go live, deploy the Node app to a host that supports persistent storage and Node.js, configure the environment variables there, attach a domain you own, and enable HTTPS. Set `NODE_ENV=production` and `COOKIE_SECURE=true` only after HTTPS is configured. Ensure `projects.json` persists across restarts/deploys and is backed up. The default Express session store is memory-only and is not appropriate for a multi-instance production deployment; configure a persistent session store (such as Redis) before scaling or using multiple instances.

Do not upload `.env` or share your admin password. For a public launch, configure a persistent session store, backups, and hosting-specific security settings.


## Android Termux note
This package uses a JSON file for project storage instead of `better-sqlite3`, because native SQLite addons may fail to compile in Android Termux. Project data is saved in `projects.json`. For a larger multi-user production deployment, use a managed database and persistent session storage.

const crypto = require('crypto');
const fs = require('fs');
const path = require('path');
const rateLimit = require('express-rate-limit');

module.exports = function setupPrivateAccess(app) {
  const file = path.join(__dirname, 'invites.json');
  const hashCode = code => crypto.createHash('sha256').update(code).digest('hex');
  const same = (a, b) => {
    const x = Buffer.from(String(a)); const y = Buffer.from(String(b));
    return x.length === y.length && crypto.timingSafeEqual(x, y);
  };
  const read = () => {
    try { const data = JSON.parse(fs.readFileSync(file, 'utf8')); if (!Array.isArray(data)) throw new Error('Invalid invites file'); return data; }
    catch (e) { if (e.code !== 'ENOENT') throw e; fs.writeFileSync(file, '[]\n', { mode: 0o600, flag: 'wx' }); return []; }
  };
  const write = items => { const temp = file + '.tmp'; fs.writeFileSync(temp, JSON.stringify(items, null, 2), { mode: 0o600 }); fs.renameSync(temp, file); };
  const csrfOK = req => { const token = req.get('x-csrf-token') || req.body?._csrf; return !!req.session.csrfToken && !!token && same(token, req.session.csrfToken); };
  const adminOnly = (req, res, next) => { if (!req.session.authenticated) return res.status(401).json({ error: 'ورود مدیر لازم است.' }); if (req.method !== 'GET' && !csrfOK(req)) return res.status(403).json({ error: 'درخواست معتبر نیست؛ صفحه را تازه‌سازی کن.' }); next(); };
  const inviteLimiter = rateLimit({ windowMs: 15 * 60 * 1000, limit: 8, standardHeaders: true, legacyHeaders: false, message: 'تلاش‌های زیادی انجام شد؛ بعداً دوباره امتحان کن.' });
  app.use((req, res, next) => {
    if (req.session?.authenticated || req.session?.invited) return next();
    if (req.path === '/' || req.path === '/index.html') return res.redirect('/invite.html');
    if (['/invite.html', '/admin-login.html', '/admin-login.js', '/styles.css', '/api/bootstrap', '/api/login', '/api/invite'].includes(req.path)) return next();
    if (req.path.startsWith('/api/')) return res.status(401).json({ error: 'برای دیدن این محتوا، کد دعوت معتبر لازم است.' });
    return res.redirect('/invite.html');
  });
  app.post('/api/invite', inviteLimiter, (req, res, next) => {
    const code = typeof req.body.code === 'string' ? req.body.code.trim() : '';
    if (!code || code.length > 100) return res.status(401).send('کد دعوت معتبر نیست یا قبلاً استفاده شده است. <a href="/invite.html">بازگشت</a>');
    const items = read(); const hash = hashCode(code);
    const item = items.find(x => !x.revoked && !x.usedAt && same(x.hash, hash));
    if (!item) return res.status(401).send('کد دعوت معتبر نیست یا قبلاً استفاده شده است. <a href="/invite.html">بازگشت</a>');
    item.usedAt = new Date().toISOString(); write(items);
    req.session.regenerate(err => { if (err) return next(err); req.session.invited = true; req.session.save(err2 => { if (err2) return next(err2); res.redirect('/'); }); });
  });
  app.get('/api/invites', adminOnly, (req, res) => { res.set('Cache-Control', 'no-store'); res.json(read().map(({ id, createdAt, usedAt, revoked }) => ({ id, createdAt, usedAt, revoked }))); });
  app.post('/api/invites', adminOnly, (req, res) => { const items = read(); const code = crypto.randomBytes(12).toString('base64url'); const item = { id: crypto.randomBytes(8).toString('hex'), hash: hashCode(code), createdAt: new Date().toISOString(), usedAt: null, revoked: false }; items.push(item); write(items); res.status(201).json({ code, invite: { id: item.id, createdAt: item.createdAt, usedAt: null, revoked: false } }); });
  app.delete('/api/invites/:id', adminOnly, (req, res) => { const items = read(); const item = items.find(x => x.id === req.params.id); if (!item) return res.status(404).json({ error: 'کد پیدا نشد.' }); item.revoked = true; write(items); res.json({ ok: true }); });
};
